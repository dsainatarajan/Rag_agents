class PlanExecuteState(TypedDict):
    plan: str
    code: str
    valid: bool
    status: str
	
def planner(state: StateGraph):
	state.update({"plan": "my plan"})
	return state

def coder(state: StateGraph):
	code = llm.invoke(state.plan + "Write the code for above plan")
	state.update({"code": code})
	return state
	
def validation(state: StateGraph):
	try:
		exec(state.code)
		state.update({"valid": True})
		return state
	exception:
		state.update({"valid": False})
		return state

def logical_check_llm(state: StateGraph):

				else if the code is logically wrong, then wrtie fix""")
	state.update({"status": status})
	return state
